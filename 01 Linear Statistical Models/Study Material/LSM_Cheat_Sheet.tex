\sect{10. ONE-WAY ANOVA \pyq{G --- NUMERICAL}}
$Y_{ij}{=}\mu{+}\alpha_i{+}\varepsilon_{ij}{=}\mu_i{+}\varepsilon_{ij}$, $i\le r$, $j\le n_i$, $n{=}\sum n_i$; $\rho(\bX){=}r$ (not $r{+}1$); $\Nsp(\bX){=}\langle(1,-1,\dots,-1)^T\rangle$.
